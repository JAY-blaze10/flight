CREATE TABLE Lib_Audit (
    Granth_No INT,
    Title VARCHAR(100),
    Author VARCHAR(100),
    Publisher VARCHAR(100),
    Count INT,
    Date DATE,
    Status VARCHAR(20)
);

\d Lib_Audit

CREATE OR REPLACE FUNCTION audit_delete()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO Lib_Audit
    VALUES (
        OLD.Granth_No,
        OLD.Title,
        OLD.Author,
        OLD.Publisher,
        OLD.Count,
        CURRENT_DATE,
        'deleted'
    );

    RETURN OLD;
END;
$$ LANGUAGE plpgsql;


CREATE TRIGGER before_delete_granth
BEFORE DELETE ON Granth
FOR EACH ROW
EXECUTE FUNCTION audit_delete();

//Create AFTER UPDATE trigger

CREATE OR REPLACE FUNCTION audit_update()
RETURNS TRIGGER AS $$
BEGIN
    INSERT INTO Lib_Audit
    VALUES (
        NEW.Granth_No,
        NEW.Title,
        NEW.Author,
        NEW.Publisher,
        NEW.Count,
        CURRENT_DATE,
        'updated'
    );

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;


//Now create the trigger:

CREATE TRIGGER after_update_granth
AFTER UPDATE ON Granth
FOR EACH ROW
EXECUTE FUNCTION audit_update();
