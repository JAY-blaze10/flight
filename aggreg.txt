Find the articles having minimum likes and maximum likes among those written by Isha
db.tech_articles.aggregate([
    {$match: {Writer: "Isha"}},
    {
        $group: {
            _id: null,
            MinLikes: {$min: "$LikesCount"},
            MaxLikes: {$max: "$LikesCount"}
        }
    }
])

Find the average number of likes of articles written by Isha
db.tech_articles.aggregate([
    {$match: {Writer: "Isha"}},
    {
        $group: {
            _id: null,
            AverageLikes: {$avg: "$LikesCount"}
        }
    }
])

Find the first and last article written by Isha
db.tech_articles.aggregate([
    {$match: {Writer: "Isha"}},
    {$sort: {_id: 1}},
    {
        $group: {
            _id: "$Writer",
            FirstArticle: {$first: "$ArticleTitle"},
            LastArticle: {$last: "$ArticleTitle"}
        }
    }
])
