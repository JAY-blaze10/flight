CREATE DATABASE CollegeDB;
USE CollegeDB;

//Create tables
 //1. Create Pupil table

CREATE TABLE Pupil (
    pupilid INT PRIMARY KEY,
    pupilname VARCHAR(50),
    guideid INT,
    pupilcity VARCHAR(50)
);

//2. Create Guide table

CREATE TABLE Guide (
    guideid INT PRIMARY KEY,
    guidename VARCHAR(50),
    guidecity VARCHAR(50),
    specialization VARCHAR(50)
);

//1. Find the guide of each pupil.
Use LEFT JOIN.

SELECT p.pupilid, p.pupilname, g.guidename,
       g.specialization
FROM Pupil p
LEFT JOIN Guide g
ON p.guideid = g.guideid;

//2. Find pupils who do not have any guide.
Use LEFT JOIN + IS NULL.

SELECT p.pupilid, p.pupilname
FROM Pupil p
LEFT JOIN Guide g
ON p.guideid = g.guideid
WHERE g.guideid IS NULL;


//Q3. Find pupils who do not have any guide as well as guides who do not have any pupil.
Use FULL OUTER JOIN.

SELECT p.pupilid, p.pupilname, g.guideid, g.guidename
FROM Pupil p
LEFT JOIN Guide g
ON p.guideid = g.guideid
WHERE g.guideid IS NULL

UNION

SELECT p.pupilid, p.pupilname, g.guideid, g.guidename
FROM Pupil p
RIGHT JOIN Guide g
ON p.guideid = g.guideid
WHERE p.pupilid IS NULL;

//Q4. Find pupils whose guide's specialization is Computer.

SELECT p.pupilid, p.pupilname, g.guidename,
       g.specialization
FROM Pupil p
INNER JOIN Guide g
ON p.guideid = g.guideid
WHERE g.specialization = 'Computer';

//Q5. Create a view containing the total number of pupils whose guide belongs to Pune.

CREATE VIEW Pune_Guide_Count AS
SELECT COUNT(p.pupilid) AS Total_Pupils
FROM Pupil p
INNER JOIN Guide g
ON p.guideid = g.guideid
WHERE g.guidecity = 'Pune';
