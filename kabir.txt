Display all documents whose TITLE is Node Basics
db.digital_library.find({TITLE: "Node Basics"})

Display documents written by Kabir OR whose TITLE is Node Basics
db.digital_library.find({
    $or: [
        {AUTHOR: "Kabir"},
        {TITLE: "Node Basics"}
    ]
})

Display documents whose TITLE is Node Basics AND AUTHOR is Kabir
db.digital_library.find({
    TITLE: "Node Basics",
    AUTHOR: "Kabir"
})

Display documents whose LIKES are greater than 20
db.digital_library.find({
    LIKES: {$gt: 20}
})

Display documents whose LIKES are greater than 100 AND whose TITLE is Node Basics OR AUTHOR is Kabir
db.digital_library.find({
    LIKES: {$gt: 100},
    $or: [
        {TITLE: "Node Basics"},
        {AUTHOR: "Kabir"}
    ]
})

Update the TITLE of Node Basics
db.digital_library.updateOne(
    {TITLE: "Node Basics"},
    {$set: {TITLE: "Node Basics Overview"}}
)

})

Display exactly two documents written by Kabir
