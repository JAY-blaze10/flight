Display documents whose TITLE is VueJS Intro

    TITLE: "VueJS Intro"
})

Display documents written by Isha OR whose TITLE is VueJS Intro
db.e_library.find({
    $or: [
        {AUTHOR: "Isha"},
        {TITLE: "VueJS Intro"}
    ]
})

Display documents whose TITLE is VueJS Intro AND AUTHOR is Isha
db.e_library.find({
    TITLE: "VueJS Intro",
    AUTHOR: "Isha"
})

Display documents whose LIKES are greater than 90
db.e_library.find({
    LIKES: {$gt: 90}
})

Display documents whose LIKES are greater than 90 AND whose TITLE is VueJS Intro OR AUTHOR is Nikhil
db.e_library.find({
    LIKES: {$gt: 90},
    $or: [
        {TITLE: "VueJS Intro"},
        {AUTHOR: "Nikhil"}
    ]
})

Update the TITLE of VueJS Intro
db.e_library.updateOne(
    {TITLE: "VueJS Intro"},
    {$set: {TITLE: "VueJS Overview"}}
)

})

Display exactly two documents written by Isha

    AUTHOR: "Isha"
